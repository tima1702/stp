# Теоретическая часть

## Git Hooks: pre-commit

**Git hook** - это скрипт, который автоматически запускается при определенных событиях в Git.

**pre-commit hook** выполняется непосредственно перед созданием коммита и может:

- Анализировать изменения в коде
- Проверять качество кода
- Запускать автоматические проверки
- Блокировать коммит при критических ошибках (опционально)

### Расположение и настройка:

```bash
# Путь к hook-файлу
.git/hooks/pre-commit

# Права на выполнение (обязательно!)
chmod +x .git/hooks/pre-commit

# Коды возврата:
# 0 - успех, коммит разрешен
# любой другой - ошибка, коммит заблокирован
```

## Проблемы именования переменных

### 🏷️ **Плохие названия переменных:**

**Одинарные буквы:**

- `a`, `b`, `c`, `x`, `y`, `z` (кроме циклов и математических формул)

**Неинформативные названия:**

- `data`, `temp`, `var`, `result`, `item`, `value`

**Сокращения без объяснений:**

- `usr` вместо `user`
- `calc` вместо `calculator`
- `btn` вместо `button`
- `mgr` вместо `manager`

### ✅ **Хорошие названия переменных:**

- `user_name`, `student_count`, `total_price`
- `is_valid`, `has_permission`, `can_edit`
- `calculate_total()`, `get_user_data()`, `validate_input()`

## Ollama API для анализа

### Базовая структура запроса:

```bash
curl -X POST http://localhost:11434/api/generate \
  -H "Content-Type: application/json" \
  -d '{
    "model": "deepseek-coder:1.3b",
    "prompt": "your prompt here",
    "stream": false,
    "options": {
      "num_predict": 100,
      "temperature": 0.1
    }
  }'
```

### Параметры:

- `num_predict` - максимальное количество токенов в ответе
- `temperature` - уровень креативности (0.0-1.0)
- `stream` - потоковая передача ответа (true/false)

## 2. Создание pre-commit hook

Создайте файл `.git/hooks/pre-commit` со следующим содержимым:

```bash
#!/bin/bash

  

# Simple pre-commit hook for checking variable names

# Checks only one thing: quality of variable naming

  

# Colors for output

GREEN='\033[0;32m'

YELLOW='\033[1;33m'

BLUE='\033[0;34m'

RED='\033[0;31m'

NC='\033[0m'

  

# Settings

OLLAMA_URL="http://localhost:11434"

MODEL="deepseek-coder:1.3b"

LOG_FILE=".git/hooks/variable-check.log"

  

echo -e "${BLUE}🔍 Checking variable names...${NC}"

  

# Logging function

log_message() {

echo "$(date '+%Y-%m-%d %H:%M:%S') $1" >> "$LOG_FILE"

}

  

# Check if Ollama is available

check_ollama() {

if ! curl -s "$OLLAMA_URL/api/tags" >/dev/null 2>&1; then

echo -e "${YELLOW}⚠️ Ollama not available, skipping check${NC}"

log_message "WARNING: Ollama not available"

exit 0

fi

log_message "INFO: Ollama is available"

}

  

# Get list of Python files for commit

get_python_files() {

git diff --staged --name-only | grep '\.py$'

}

  

# Check variable names in file

check_variable_names() {

local file="$1"

echo -e "${BLUE}📄 Checking file: $file${NC}"

log_message "INFO: Checking file $file"

# Read only first 30 lines of file for simplicity

local file_content

file_content=$(head -n 30 "$file" 2>/dev/null)

if [ -z "$file_content" ]; then

echo -e "${YELLOW}⚠️ File is empty or not accessible${NC}"

log_message "WARNING: File $file is empty or not accessible"

return 0

fi

# Simple English prompt about variable names

local prompt="Find bad variable names in this Python code: $file_content. Look for variables with bad names: single letters (a, b, c, x, y, z), uninformative names (data, temp, var, result, item), abbreviations (usr, calc, btn, mgr). Answer with a list of bad variables or write 'good' if no problems found."

# Escape quotes and newlines for JSON

local escaped_prompt

escaped_prompt=$(echo "$prompt" | sed 's/"/\\"/g' | tr '\n' ' ')

log_message "INFO: Sending request to LLM for $file"

echo -e "${BLUE}🔄 Sending request to Ollama...${NC}"

local response

response=$(curl -s -m 30 -X POST "$OLLAMA_URL/api/generate" \

-H "Content-Type: application/json" \

-d "{

\"model\": \"$MODEL\",

\"prompt\": \"$escaped_prompt\",

\"stream\": false,

\"options\": {

\"num_predict\": 700,

\"temperature\": 0.1

}

}")

local curl_exit_code=$?

# Debug information

echo -e "${BLUE}🔍 Curl exit code: $curl_exit_code${NC}"

echo -e "${BLUE}🔍 Response size: ${#response} characters${NC}"

log_message "INFO: Curl exit code: $curl_exit_code, Response size: ${#response}"

if [ $curl_exit_code -ne 0 ]; then

echo -e "${RED}❌ Curl error (code: $curl_exit_code)${NC}"

log_message "ERROR: Curl failed with code $curl_exit_code"

return 1

fi

if [ -z "$response" ]; then

echo -e "${RED}❌ Empty response from Ollama${NC}"

log_message "ERROR: Empty response from Ollama"

return 1

fi

# Log full response for debugging

log_message "RESPONSE: $response"

# Show first 150 characters of response for debugging

echo -e "${BLUE}🔍 Response start: ${response:0:150}...${NC}"

# Extract text response using jq (more reliable than python)

local llm_response

if command -v jq >/dev/null 2>&1; then

llm_response=$(echo "$response" | jq -r '.response' 2>/dev/null)

local jq_exit_code=$?

echo -e "${BLUE}🔍 jq exit code: $jq_exit_code${NC}"

log_message "INFO: jq exit code: $jq_exit_code"

else

# Fallback to python if jq not available

llm_response=$(echo "$response" | python3 -c "import sys, json; print(json.load(sys.stdin)['response'])" 2>/dev/null)

local python_exit_code=$?

echo -e "${BLUE}🔍 Python exit code: $python_exit_code${NC}"

log_message "INFO: Python exit code: $python_exit_code"

fi

if [ -z "$llm_response" ] || [ "$llm_response" = "null" ]; then

echo -e "${RED}❌ Could not extract LLM response${NC}"

echo -e "${BLUE}🔍 Full response: $response${NC}"

log_message "ERROR: Could not extract LLM response"

return 1

fi

# Log the LLM response

log_message "LLM_RESPONSE: $llm_response"

echo -e "${BLUE}🔍 Full LLM response: $llm_response${NC}"

# Show result

echo "┌─────────────────────────────────────────────┐"

echo "│ 🏷️ Variable Names Analysis"

echo "└─────────────────────────────────────────────┘"

# Check if response indicates good naming

if echo "$llm_response" | grep -iq "good\|no problems\|no issues\|no bad\|все хорошо\|хорошо"; then

echo -e "${GREEN}✅ Variable names are good!${NC}"

log_message "RESULT: Variable names are good for $file"

return 0

else

echo -e "${YELLOW}⚠️ Issues found: $llm_response${NC}"

echo -e "${BLUE}💡 Consider improving variable names${NC}"

log_message "RESULT: Issues found in $file: $llm_response"

return 1

fi

}

  

# Main function

main() {

log_message "=== Starting variable name check ==="

check_ollama

# Get list of Python files

python_files=$(get_python_files)

if [ -z "$python_files" ]; then

echo -e "${GREEN}✅ No Python files to check${NC}"

log_message "INFO: No Python files to check"

exit 0

fi

echo -e "${BLUE}📁 Found Python files:${NC}"

echo "$python_files"

log_message "INFO: Found files: $python_files"

echo ""

# Check each file

local total_files=0

local files_with_issues=0

for file in $python_files; do

if [ -f "$file" ]; then

total_files=$((total_files + 1))

if ! check_variable_names "$file"; then

files_with_issues=$((files_with_issues + 1))

fi

echo ""

fi

done

# Final statistics

echo -e "${BLUE}📊 Summary:${NC}"

echo " 📄 Files checked: $total_files"

if [ $files_with_issues -gt 0 ]; then

echo -e " ⚠️ Files with issues: ${YELLOW}$files_with_issues${NC}"

else

echo -e " ✨ All files are good!"

fi

log_message "SUMMARY: Files checked: $total_files, Files with issues: $files_with_issues"

log_message "=== Variable name check completed ==="

# Always allow commit (just warn)

exit 0

}

  

# Run the check

main
```


```python
import math

# Хорошие названия переменных
def calculate_circle_area(radius, height, coefficient):
    base_area = radius + height
    scaled_area = base_area * coefficient
    half_area = scaled_area / 2
    final_area = half_area * math.pi
    return final_area

def process_user_scores(user_scores):
    doubled_scores = []
    for score in user_scores:
        doubled_value = score * 2
        doubled_scores.append(doubled_value)
    return doubled_scores

class ScoreManager:
    def __init__(self):
        self.scores = []
        self.current_user = None
        self.total_count = 0
        self.average_score = 0
    
    def calculate_user_average(self, user_scores, total_games):
        sum_scores = user_scores + total_games
        user_average = sum_scores * self.total_count
        return user_average

# Глобальные переменные с понятными названиями
```

```python
def calculate_price(base_price, tax_rate):
    # Хорошие названия
    subtotal = base_price * 1.0
    tax_amount = subtotal * tax_rate
    
    # Плохие названия
    temp = tax_amount + subtotal
    result = temp

def process_data(user_list):
    # Смешанные названия
    processed_users = []
```
